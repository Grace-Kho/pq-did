cmake_minimum_required(VERSION 3.20)
project(pqdid_environment LANGUAGES C CXX)
find_package(liboqs 0.16.0 EXACT CONFIG REQUIRED)
add_executable(toolchain_smoke toolchain_smoke.cpp toolchain_probe.c)
target_compile_features(toolchain_smoke PRIVATE c_std_11 cxx_std_17)
target_compile_options(toolchain_smoke PRIVATE -Wall -Wextra -Werror)
target_link_libraries(toolchain_smoke PRIVATE OQS::oqs)
enable_testing()
add_test(NAME native_toolchain_and_liboqs COMMAND toolchain_smoke)
